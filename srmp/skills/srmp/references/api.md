# === Planner selection and planner_context ===
# make_planner(articulation_names, planner_context): all planner_context values are strings;
# keys and planner names are case-sensitive. print_available_planners() lists what this build has.
#   planner_id: single-agent "Astar" / "wAstar" / "ARAstar" / "MHAstar" / "wPASE" / "MGS" / "vamp_rrtc"
#               (vamp_rrtc: VAMP RRT-Connect, joint-space goals, needs the VAMP backend);
#               multi-agent (len(articulation_names) > 1) "ECBS" / "xECBS"
#   MGS only: heuristic (root-to-root, default "joint_steps"; not "bfs"),
#             anchor_heuristic (default "bfs"), g_num (graphs, default "10"), weight (default "100"),
#             anchor_search ("true"/"false", default "true"), root_ik_threads (default: half the hardware threads),
#             anchor_orientation_weight (cost/rad of EE orientation error in the anchor heuristic, default "0"),
#             expansion_threads (edge checks per expansion, coal only; default: all cores),
#             snap_ik_seeds (concurrent IK seeds per snap attempt on pose goals, default "1"),
#             graph_growth ("on_stall" default / "upfront"), stall_window (default "50"), root_interval ("0"),
#             root_selection ("frontier_nearest" default / "sequential")
#   heuristic_<articulation>, mprim_path_<articulation>  (per-agent, multi-agent only)
#   weight_low_level_heuristic (applies "50.0" when omitted, though it logs 1.0),
#   high_level_focal_suboptimality (default 1.3),
#   low_level_focal_suboptimality (default 1.3)
#   verbose ("true"/"1" for search progress diagnostics, default "false")
#   validity_backend: "coal" (default) / "vamp" (raises if the robot has no matching module) /
#               "auto" (VAMP when every planned robot supports it, else coal; never raises)
#   VAMP scene translation (meters unless noted): mesh_sampling_resolution (default "0.01"),
#     pointcloud_point_radius ("0.01"), articulation_link_sampling_resolution ("0.05"),
#     attachment_spherization_resolution ("0.05"), attachment_max_spheres ("512"; an attached
#     mesh over the budget raises -- raise it, or attach a primitive like add_box instead)

# === Kinematics: ee_pose and plan_screw ===
# compute_ik(articulation_name, ee_pose, init_state_val): ee_pose is a Pose or a flat
#   [x, y, z, roll, pitch, yaw] list, in the WORLD frame (the frame compute_fk returns; the robot's
#   base pose is applied).
# plan_screw(articulation_name, link_name, ...): Cartesian screw-motion plan via Jacobian
#   resolved-rate stepping (no IK solve); each step is collision- and joint-limit-checked.
#   Give exactly one of:
#     - end_pose: target pose for link_name (axis, pitch and angle are derived)
#     - axis_direction + angle (+ optional axis_point, pitch): explicit screw motion, e.g. a
#       hinge or door swing. axis_point defaults to link_name's current position.
#   start_qpos defaults to the articulation's current qpos and is also set as a side effect.
#   Raises RuntimeError on inconsistent parameters, zero relative rotation, collision, a joint-limit
#   violation, a singularity stall, or exceeding max_steps.

# === Collision backends ===
# is_state_colliding(...) returns True for a COLLISION. is_vamp_state_valid(...) is the VAMP
#   counterpart and has the opposite polarity: True means a VALID configuration.
# is_robot_vamp_supported(name) is False also when a VAMP module of that name exists but does not
#   match this robot's kinematics (e.g. VAMP's built-in ur5 vs the registry ur5); that robot then
#   plans on coal.

# === Visualizer ===
# start_visualizer(type="viser", **kwargs): the preferred way to get a live 3D viewer. It constructs
#   and attaches the viewer, opens the browser, and returns the instance. type: "viser" or "meshcat".
#   If a viser viewer is already attached and running, that one is returned (pass port=... only to
#   deliberately open a second viewer). e.g. viz = planner.start_visualizer("viser")
#   The viser viewer exposes:
#     viz.animate_trajectory(trajectories, dt=0.05, robot_name=None, fps=30.0) -> None
#     viz.stop() -> None
# get_visualizer(index=0): an already-attached visualizer, or None.
# attach_visualizer(listener) / detach_visualizer(listener): low-level, manually attach or detach a
#   VisualizerListener.

# === VAMP plugins ===
# VAMP modules for more robots, without recompiling srmp. add_robot("<registry name>") installs and
